# freebsd-config — pkg(8) package set for FreeBSD 15.1 (amd64/aarch64).
# Install with:  sudo pkg install $(grep -v '^[[:space:]]*#' packages.txt | grep -v '^[[:space:]]*$' | tr '\n' ' ')
# Lines starting with # are comments. Entries marked OPTIONAL may fail on
# quarterly branches — the installer continues past missing optionals.
#
# FreeBSD naming convention: all names are lowercase pkg names
# (ports origin shown in trailing comments). Linux-only Void packages have
# been replaced: NetworkManager -> native wlan(4)+wpa_supplicant(8),
# ufw/nftables -> pf(4), cronie -> base cron(8), chrony -> base ntpd(8),
# elogind -> seatd, alsa-* -> native oss(4)/mixer(8)+optional pipewire,
# power-profiles-daemon -> base powerd(8).

swaylock             # x11/swaylock

# NOTE — installed separately by install.sh (not via pkg):
#   matugen     — `cargo install matugen` (no FreeBSD port; theming engine)
#   awww        — `cargo install awww` (no FreeBSD port; falls back to swaybg)
#   autotiling  — `pip install autotiling` / `pkg install py-autotiling` if present
#   rofimoji    — `pip install rofimoji` (emoji picker backend)
#   rofi-calc   — build from ports if available, else skip (calc menu degrades gracefully)
#   voxtype     — Linux-only dictation; skipped on FreeBSD (binds become no-ops)
#   power-profiles-daemon / powerprofilesctl — Linux-only; replaced by base powerd(8)
#   NetworkManager/nmtui — replaced by native wlan(4)+wpa_supplicant(8); nm-* kept as
#                          optional manual installs only
